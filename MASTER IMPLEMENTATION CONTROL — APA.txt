1.3 BillingVersion (ROOM + MONTH LEVEL)
model BillingVersion {
  id              String   @id @default(cuid())

  billingMonthId  String
  billingMonth    BillingMonth @relation(fields: [billingMonthId], references: [id], onDelete: Cascade)

  roomId          String
  room            Room @relation(fields: [roomId], references: [id], onDelete: Cascade)

  versionNo       Int
  isActive        Boolean @default(true)

  snapshotData    Json
  totalAmount     Decimal

  revertedFromId  String?
  revertedFrom    BillingVersion? @relation("RevertChain", fields: [revertedFromId], references: [id])
  revertedTo      BillingVersion[] @relation("RevertChain")

  createdAt       DateTime @default(now())
  createdBy       String

  documents       DocumentVersion[]

  @@unique([billingMonthId, roomId, versionNo])
  @@index([billingMonthId, roomId, isActive])
}

RULES:

Only ONE active per room per month

Restore = create new version

Never update old snapshot
